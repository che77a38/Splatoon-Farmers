数值选项: 进图时间,6,0,99

数值选项: 武器删除轮数,1,0,300

列表选项: 手柄唤醒初始,是,否

$总时长 = 一轮时间鲑鱼卵为准

$配件RCD = R配件CD若0为自动
$配件ACD = A配件CD若0为自动
$跑图次数 = 0
$删除计数 = 0
$轮数 = 武器删除轮数
$手动模式标志 =7
IF $配件LCD == 0

ENDIF
IF $配件RCD == 0

ENDIF
IF $配件ACD == 0

ENDIF
$计数器 = 0
IF 手柄唤醒初始 == 是

ENDIF
WAIT 1000

    WAIT 500
    $跑图次数 +=1
    $删除计数 +=1

    WAIT 2000

    WAIT 1000
    IF $删除计数   == $轮数
        IF 武器删除 == 仅白色
            CALL 武器删除白
        ELIF 武器删除 == 绿及以下
            CALL 武器删除白绿
        ELIF 武器删除 == 紫及以下
            CALL 武器删除白绿紫
        ELIF 武器删除 == 金及以下
            CALL 武器删除白绿紫金
        ENDIF
        $删除计数 = 0
    ENDIF
NEXT

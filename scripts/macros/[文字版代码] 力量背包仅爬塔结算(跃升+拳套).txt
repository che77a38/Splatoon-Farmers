标题: 杏棱力量背包刷配件
数值选项: 进图时间,6,0,99
数值选项: 出图时间,9,0,99
数值选项: 武器删除轮数,1,0,300
列表选项: 武器删除,无,仅白色,绿及以下,紫及以下,金及以下
列表选项: 手柄唤醒初始,是,否

$跑图次数 = 0
$删除计数 = 0
$轮数 = 武器删除轮数
IF 手柄唤醒初始 == 是
    CALL 手柄唤醒
ENDIF
WAIT 1000
FOR
    WAIT 500
    $跑图次数 +=1
    $删除计数 +=1
    PRINT "第" $跑图次数 "跑图"
    X 50
    WAIT 500
    A 50
    WAIT 500
    A 50
    WAIT 50
    A 50
    WAIT 1000
    A 50
    WAIT 进图时间*1000
    CALL 图内跑步
    PLUS 50
    WAIT 100
    DOWN 50
    WAIT 100
    A 50
    WAIT 500
    RIGHT 50
    WAIT 100
    A 50
    WAIT 出图时间*1000
    WAIT 900
    A 50
    WAIT 900
    A 50
    WAIT 900
    A 50
    WAIT 2000
    A 50
    WAIT 900
    A 50
    WAIT 1000
    IF $删除计数   == $轮数 
        IF 武器删除 == 仅白色
            CALL 武器删除白
        ELIF 武器删除 == 绿及以下
            CALL 武器删除白绿
         ELIF 武器删除 == 紫及以下
            CALL 武器删除白绿紫
         ELIF 武器删除 == 金及以下
            CALL 武器删除白绿紫金
        ENDIF
        $删除计数 = 0
    ENDIF
NEXT
